# How sb's parser (cli-core's YAML pre-pass, then saphyr-parser 0.3.3 and
# sb's own scalar resolution) differs from libyaml (expected.json) on the
# corpus. Asserted exactly by tests/yaml_corpus.rs; a line here is a known
# difference, not a target. Each is in the config spec's Differences list.
# The pre-pass closed config-H1 and config-L1 (decision 28).

# Since decision 38 sb refuses every case libyaml refuses (a second
# document among them), so no line here reads "sb accepts".

# Line breaks: NEL, LS and PS are line breaks to libyaml and text to
# saphyr, so sb refuses a file holding one (docs/specs/config.md), even
# where libyaml reads it.
ls-line-break: libyaml accepts, sb refuses: syntax error: U+2028 (LS) is a line break to Ansible's YAML parser and not to sb's (in a double-quoted string, write it as \u2028) at line 1, column 5

# Resolution: sb resolves plain scalars as yaml.v3 does (YAML 1.2 core,
# as sb-go; docs/specs/config.md), libyaml's PyYAML resolver is YAML 1.1.
# The schema rules read yes/no/on/off as Ansible booleans either way.
saltbox-accounts: plex.tfa: resolves to str "no", libyaml bool "false"
saltbox-colon-in-plain: time: resolves to str "12:30", libyaml int "750"
saltbox-numbers: big: resolves to float "1e20", libyaml int "99999999999999999999"
saltbox-numbers: exp: resolves to float "1000.0", libyaml str "1e3"
saltbox-yes-no: dns.ipv4: resolves to str "yes", libyaml bool "true"
saltbox-yes-no: dns.ipv6: resolves to str "no", libyaml bool "false"
saltbox-yes-no: dns.other: resolves to str "off", libyaml bool "false"
saltbox-yes-no: dns.proxied: resolves to str "on", libyaml bool "true"

# A top-level scalar: libyaml reads it, and sb refuses it as a config
# file, which must be a mapping of keys (the YAML read agrees: cli-core's
# unit tests check these two scalars' values against libyaml's).
top-level-block-scalar-blank-then-comment: libyaml accepts, sb refuses: the file must be a mapping of keys, not a scalar at line 1, column 5
top-level-block-scalar-then-comment: libyaml accepts, sb refuses: the file must be a mapping of keys, not a scalar at line 1, column 5

# A top-level block scalar with no content, whose blank lines hold
# spaces, before an end marker: saphyr refuses it where libyaml reads an
# empty scalar. A refusal, not a different value; no Saltbox file has a
# top-level scalar.
top-level-block-scalar-spaces-before-end: libyaml accepts, sb refuses: syntax error: wrongly indented line in block scalar at line 3, column 1
