# How sb's parser (saphyr-parser 0.3.3, sb's own scalar resolution)
# differs from libyaml (expected.json) on the corpus. Asserted exactly by
# tests/yaml_corpus.rs; a line here is a known difference, not a target.

# config-H1: layouts libyaml (and so Ansible) accepts and sb refuses.
double-quoted-multi-line-close-at-column-0: libyaml accepts, sb refuses: syntax error: invalid indentation in quoted scalar at line 1, column 7
flow-map-empty-split: libyaml accepts, sb refuses: syntax error: invalid indentation at line 2, column 1
flow-seq-close-at-key-column-after-quoted: libyaml accepts, sb refuses: syntax error: invalid indentation at line 3, column 1
flow-seq-close-left-of-key-column: libyaml accepts, sb refuses: syntax error: invalid indentation at line 4, column 1
flow-seq-empty-split: libyaml accepts, sb refuses: syntax error: invalid indentation at line 2, column 1
flow-seq-items-one-per-line: libyaml accepts, sb refuses: syntax error: invalid indentation at line 4, column 1
saltbox-flow-list-split-close-at-key: libyaml accepts, sb refuses: syntax error: invalid indentation at line 5, column 3
tab-after-colon-in-nested: libyaml accepts, sb refuses: syntax error: ':' must be followed by a valid YAML whitespace at line 2, column 8
tab-after-colon: libyaml accepts, sb refuses: syntax error: ':' must be followed by a valid YAML whitespace at line 1, column 6

# config-L1: tab layouts libyaml refuses and sb accepts (the tab after a
# dash found by this corpus).
tab-after-dash: libyaml refuses (found character that cannot start any token at line 2, column 4), sb accepts
tab-indented-top-level: libyaml refuses (found character that cannot start any token at line 1, column 1), sb accepts
tab-only-line: libyaml refuses (found a tab character that violates indentation at line 2, column 1), sb accepts

# Documents: sb reads the first document only, as yaml.v3 and sb-go
# (docs/specs/config.md); Ansible's loader wants a single document.
multi-document: libyaml refuses (but found another document at line 2, column 1), sb accepts

# Resolution: sb resolves plain scalars as yaml.v3 does (YAML 1.2 core,
# as sb-go; docs/specs/config.md), libyaml's PyYAML resolver is YAML 1.1.
# The schema rules read yes/no/on/off as Ansible booleans either way.
saltbox-accounts: plex.tfa: resolves to str "no", libyaml bool "false"
saltbox-colon-in-plain: time: resolves to str "12:30", libyaml int "750"
saltbox-numbers: big: resolves to float "1e20", libyaml int "99999999999999999999"
saltbox-numbers: exp: resolves to float "1000.0", libyaml str "1e3"
saltbox-yes-no: dns.ipv4: resolves to str "yes", libyaml bool "true"
saltbox-yes-no: dns.ipv6: resolves to str "no", libyaml bool "false"
saltbox-yes-no: dns.other: resolves to str "off", libyaml bool "false"
saltbox-yes-no: dns.proxied: resolves to str "on", libyaml bool "true"
